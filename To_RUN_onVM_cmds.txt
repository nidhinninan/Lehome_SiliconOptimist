# === GPU memory cleanup (run after a crashed/killed training run) ===
```python

# 1. Check what's holding GPU memory
nvidia-smi

# 2. Kill all processes still occupying /dev/nvidia* devices (catches zombie workers)
sudo fuser -k /dev/nvidia*

# 3. Also nuke any lingering Python training processes by name
pkill -9 -f lerobot_train_with_plugins || true
pkill -9 -f lerobot_train              || true
pkill -9 -f "python -m lerobot"        || true

# 3. If zombie compute apps are still stuck in VRAM, extract their PIDs directly from nvidia-smi and kill them:
# (This is safer than a blanket fuser -k /dev/nvidia*)
sudo kill -9 $(nvidia-smi --query-compute-apps=pid --format=csv,noheader) || true

# 4. Verify GPU memory is freed (MiB used should drop to near 0%)
sleep 2 && nvidia-smi --query-gpu=memory.used,memory.free --format=csv
```
